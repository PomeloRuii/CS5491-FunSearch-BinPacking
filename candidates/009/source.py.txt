def priority_v1(item: float, bins: np.ndarray) -> np.ndarray:
    bins = np.asarray(bins, dtype=float)
    residual = bins - item
    valid = residual >= 0
    if not np.any(valid):
        return np.zeros_like(bins)
    valid_residual = residual[valid]
    valid_bins = bins[valid]
    mean_residual = np.mean(valid_residual)
    std_residual = np.std(valid_residual)
    if std_residual < 1e-12:
        z_scores = np.zeros_like(valid_residual)
    else:
        z_scores = (valid_residual - mean_residual) / std_residual
    item_ratio = item / np.where(valid_bins > 0, valid_bins, 1.0)
    waste_penalty = item_ratio * 0.5
    scores = z_scores - waste_penalty
    result = np.full_like(bins, -1e18)
    result[valid] = scores
    return result
