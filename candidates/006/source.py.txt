def priority_v1(item: float, bins: np.ndarray) -> np.ndarray:
    bins = np.asarray(bins, dtype=float)
    residual = bins - item
    used = residual > 0
    if np.any(used):
        min_used = np.min(residual[used])
        scores = np.where(used, 1.0 / (residual + 1e-12), 0.0)
        scores = np.where(residual == min_used, scores + 1e6, scores)
    else:
        scores = np.where(bins > 0, 1.0 / (bins + 1e-12), 0.0)
    return scores
