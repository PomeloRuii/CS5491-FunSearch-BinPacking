def priority_v1(item: float, bins: np.ndarray) -> np.ndarray:
    bins = np.asarray(bins, dtype=float)
    residual = bins - item
    valid = residual >= 0
    residual_safe = np.where(valid, residual, 1.0)
    item_safe = max(item, 1e-12)
    ratio = item_safe / residual_safe
    log_ratio = np.log(ratio)
    penalty = np.where(valid, log_ratio, -1e18)
    return penalty
