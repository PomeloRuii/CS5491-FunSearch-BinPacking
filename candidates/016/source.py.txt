def priority_v1(item: float, bins: np.ndarray) -> np.ndarray:
    bins = np.asarray(bins, dtype=float)
    residual = bins - item
    residual = np.where(residual >= 0, residual, 0.0)
    residual = np.where(residual > 0, residual, 1e-12)
    ratio = item / residual
    ratio = np.where(ratio > 0, ratio, 1e-12)
    log_ratio = np.log(ratio)
    penalty = np.where(residual < item * 0.5, 1.0, 0.0)
    score = -log_ratio - penalty
    return score
